\begin{definition}\label{def:norma-general}
    Una \emph{norma} en $\mathbb{R}^n$ es una función $\|\cdot\| : \mathbb{R}^n \to \mathbb{R}$ que asigna a cada vector $\mathbf{x} \in \mathbb{R}^n$ un número real no negativo $\|\mathbf{x}\|$, y que satisface las siguientes propiedades para todos $\mathbf{x},\;\mathbf{y} \in \mathbb{R}^n$ y todo escalar $\lambda \in \mathbb{R}$:
    \begin{enumerate}
        \item Positividad: $\|\mathbf{x}\| \geq 0\;\land\;\|\mathbf{x}\| = 0 \iff \mathbf{x} = \boldsymbol{0}$.
        \item Homogeneidad: $\|\lambda \mathbf{x}\| = |\lambda|\|\mathbf{x}\|$.
        \item Desigualdad triangular: $\|\mathbf{x} + \mathbf{y}\| \leq \|\mathbf{x}\| + \|\mathbf{y}\|$.
    \end{enumerate}
\end{definition}

\begin{obs}
    El espacio $\Rn{n}$ existen muchas funciones que cumplen con las propiedades anteriormente mecionadas, las cuales algunas se mostrar\'an a continuaci\'on. Notar que para todas ellas miden la ``magnitud'' o ``distancia'' de un vector, aunque cada una con una geometr\'ia distinta.
\end{obs}

\begin{definition}\label{def:norm1}
    Sea $\mathbf{x}\in \mathbb{R}^n$. Se define la \emph{norma 1} de $\mathbf{x}$, denotada por $\|\mathbf{x}\|_1$, como
    \[
        \|\mathbf{x}\|_1 = \sum_{i=1}^n |x_i|,
    \]
    donde $\mathbf{x}=(x_1,\ldots,x_n)$, con $i=1,\ldots,n.$
\end{definition}

\begin{definition}\label{def:norm2}
    Sea $\mathbf{x}\in \mathbb{R}^n$. Se define la \emph{norma 2} o \emph{norma euclidiana} de $\mathbf{x}$, denotada por $\|\mathbf{x}\|_2$, como
    \[
        \|\mathbf{x}\|_2 = \sqrt{ \sum_{i=1}^n x_i^2 }.
    \]
    donde $\mathbf{x}=(x_1,\ldots,x_n)$, con $i=1,\ldots,n.$
\end{definition}

\begin{definition}
    Sean $\mathbf{x},\;\mathbf{y}\in\Rn{n}$. Se define la \emph{distancia} entre $\mathbf{x}\text{ e }\mathbf{y}$ como la funci\'on d$:\Rn{n}\times\Rn{n}\to\R$ tal que
    \[
        \text{d}(\mathbf{x},\mathbf{y})=\|\mathbf{x}-\mathbf{y}\|,
    \]
    donde $\|\cdot\|$ es una norma cualquiera de $\Rn{n}$.
\end{definition}

\begin{definition} \textbf{Bola de radio $\delta$ y centro $x_o$.} \label{def:bola}
    Sean $x_o \in \Rn{n},\;\delta > 0$ y $\text{d}$ una distancia en $\Rn{n}$. La \emph{bola de radio $\delta$ y centro $x_o$}, que denotamos con $B_{\delta}(x_o)$, es el conjunto
    \[
     B_{\delta}(x_o) = \left\lbrace x \in \Rn{n} : \text{d}(x,x_o) < \delta \right\rbrace.
    \]
    El conjunto
    \[
     B^*_{\delta}(x_o) = B_{\delta}(x_o) \smallsetminus \{ x_o \}
    \]
    es la \emph{bola reducida de radio $\delta$ y centro $x_o$}.
   
   \end{definition}

   \begin{definition} \textbf{Punto interior, interior de un conjunto.} \label{def:interior}
    Sean $A \subset \Rn{n} \text{ y } \vect{x}_0 \in A$. Decimos que $\vect{x}_0$ es \emph{punto interior} de $A$ si
    \[
     \exists\:\delta > 0: B_{\delta}(\vect{x}_0) \subset A.
    \]
    El \emph{interior} de $A$, que denotamos con $A^{\circ}$, es el conjunto de todos los puntos interiores de $A$:
    \[
    A^{\circ}= \left\lbrace\vect{x}\in A :\vect{x}\text{ es punto interior de } A \right\rbrace.
    \]
   \end{definition}
   
   
   \begin{definition} \textbf{Conjunto abierto.} \label{def:abierto}
       Sea $A \subset \Rn{n}$. Decimos que $A$ es \emph{abierto} si
   \[
        \forall \vect{x}_0 \in \exists \delta > 0 : B_{\delta}(\vect{x}_0) \subset A.\footnote{\text{\textit{i.e.}: $A$  es abierto si todos sus puntos son interiores.}}
   \]
   \end{definition}
   
   \begin{propertie}  \label{prop:abierto_int}    Sea  $A \subseteq \Rn{n}$, entonces:
   \begin{enumerate}
    \item \(A\subseteq \interior{A}\).
    \item \(A\) es abierto \(\iff A=\interior{A}\).
    \end{enumerate}
    \end{propertie}
   
   \begin{definition} \textbf{Conjunto cerrado.} \label{def:cerrado}
    Sea $A \subset \Rn{n}$. Decimos que $A$ es \emph{cerrado} si su complemento $A^c = \Rn{n} \smallsetminus A$ es abierto.
   \end{definition}
   
   \begin{definition} \textbf{Frontera de un conjunto.} \label{def:frontera}
    Sea $A \subset \Rn{n}$. La \emph{frontera} de $A$, que denotamos con $\partial A$ o $\frontera(A)$, es el conjunto
    \[
     \partial A = \left\lbrace \vect{x}_0 \in \Rn{n} \;|\; \forall \delta > 0, \;
     B_{\delta}(\vect{x}_0) \cap A \ne \varnothing\;\land\;
     B_{\delta}(\vect{x}_0) \cap A^c \ne \varnothing \right\rbrace.
    \]
   \end{definition}
   
   \begin{propertie} \label{prop:cerrado_front}
    Un conjunto $A \subset \Rn{n}$ es cerrado si y solo si $\partial A \subset A$.
   \end{propertie}
   
   \begin{definition} \textbf{Punto de acumulaci\'on.} \label{def:pto_acum}
    Sean $A \subset \Rn{n} \text{ y } \vect{x}_0 \in \Rn{n}$. Decimos que $\vect{x}_0$ es \emph{punto de acumulaci\'on} de $A$ si
    \[
     \forall \delta > 0 \left( B_{\delta}(\vect{x}_0) - \{ \vect{x}_0 \} \right) \cap A \ne \varnothing. 
    \]
    Denotamos el conjunto de todos los puntos de acumulaci\'on de $A$ con $A'$:
    \[
     A' = \{\vect{x}\in \Rn{n} :\vect{x}\text{ es punto de acumulaci\'on de } A \}.
    \]
   
   \end{definition}
   
   \begin{definition} \textbf{Entorno de un punto.} \label{def:entorno}
    Sean $N \subset  \Rn{n}$ y $\vect{x}_0 \in \Rn{n}$. Decimos que $N$ es \emph{entorno} de $\vect{x}_0$ si
    \[
     \exists \delta > 0 \;|\; B_{\delta}(\vect{x}_0) \subset N.
    \]
   \end{definition}
   
   \begin{definition} \textbf{Punto aislado.} \label{def:pto_ais}
    Sean $A \subset  \Rn{n}$ y $\vect{x}_0 \in A$. Decimos que $\vect{x}_0$ es \emph{punto aislado} de $A$ si $\vect{x}_0$ no es punto de acumulaci\'on de $A$.
   \end{definition}
   
   \begin{definition} \textbf{Conjunto acotado.} \label{def:conj_acotado}
    Sea $A \subset \Rn{n}$. Decimos que $A$ es un \emph{conjunto acotado} si existe $\delta > 0$ tal que la bola de radio $\delta$  centrada en el origen contiene al conjunto $A$, es decir, si $\exists \delta > 0 : A \subset B_{\delta}(\mathbf{0})$.\\
    Equivalentemente, $A$ es un \emph{conjunto acotado} si $\exists \delta > 0, \vect{x}_0 \in \Rn{n} : A \subset B_{\delta}(\vect{x}_0)$.
   \end{definition}
   
   \begin{definition}
    Sea un conjunto $A\subseteq\Rn{n}$. $A$ se llama \emph{conexo} si dados dos puntos cualquiera de $A$ se los puede unir por una curva continua que este incluida en $A$.
\end{definition}
