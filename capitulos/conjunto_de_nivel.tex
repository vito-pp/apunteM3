

%---------------------------------------------------------def conjunto de nivel----------------------------------------------------
\begin{definition} [Conjunto de nivel] 
    \label{def:conjunto de nivel}
     \mbox{}
     
    Sea $f: U \subset \mathbb{R}^n \rightarrow \mathbb{R}$ y sea $c \in \mathbb{R}$, se define  el conjunto de nivel de $f$ de nivel  $c$, y se nota $C(c,f)$,  a  
    \[
    C(c,f)=\{ \mathbf{x} \in U \backslash f (\mathbf{x})=c \}.
     \]
    Es decir,  $C(c,f)$ son aquellos puntos $x\in\ U$ para las cuales $f(x)=c$, o dicho de otra manera,  $C(c,f)$ es la preimagen de $c$ por $f$.   Si $n=2$ hablamos de curva de nivel  y si $n=3$  hablamos de superficie de nivel  
    
    \begin{obs}  Si $f: A \subseteq \mathbb{R}^n \rightarrow \mathbb{R}$ entonces $C(c,f)  \subset \mathbb{R}^{n}.$
       \end{obs}
\end{definition}

%---------------------------------------------------------def corte transversal---------------------------------------------------------------
    
\begin{definition}
        Se llama \textbf{corte transversal} de un conjunto \( S \subseteq \mathbb{R}^n \) a la intersección de \( S \) con un hiperplano \( H \subseteq \mathbb{R}^n \).
        
        Formalmente, dado un hiperplano \( H \), el corte transversal es el conjunto:
        \[
        S \cap H
        \]
        que describe la "sección" de \( S \) en el hiperplano \( H \).
        
        En \(\mathbb{R}^3\), típicamente, un corte transversal se realiza con un plano de la forma \( z = c \) (plano horizontal), \( x = c \) (plano vertical) o \( y = c \).
\end{definition}

\begin{example}   Sea  el campo  $f: \mathbb{R}^{2} \rightarrow \mathbb{R} \:|\:  f(x,y)=x^2+y^2.$

    1. Calcular y graficar los conjuntos de nivel de $f$.
    
    2. Hacer un gr\'afico de  $graf(f)$.

     $$C(c,f)=\{(x,y) \in U \backslash   x^2+y^2  =c \}.$$
    Tomando, para ilustrar,  $c=0$, $c=1$ y $c=4$, 
    % quedando (ver (\ref{circ})) 
     \[
    C(0,f)=\{(x,y) \in U \backslash x^2+y^2=0 \}
    \]
     \[
    C(1,f)=\{(x,y) \in U \backslash x^2+y^2=1 \}
    \]
     \[
    C(4,f)=\{(x,y) \in U \backslash x^2+y^2=4 \}
     \]

    \begin{obs}
        Dada la ecuación \( z^2 = x^2 + y^2 \), si despejamos \( z \) obtenemos dos expresiones:
        
        \[
        z = \sqrt{x^2 + y^2} ,
        \quad
        z = -\sqrt{x^2 + y^2} 
        \]
        
        Definimos entonces las funciones:
        
        \[
        f_1 : \mathbb{R}^2 \to [0,+\infty), \quad f_1(x,y) = \sqrt{x^2 + y^2},
        \]
        \[
        f_2 : \mathbb{R}^2 \to (-\infty,0], \quad f_2(x,y) = -\sqrt{x^2 + y^2}.
        \]
        
        Así, los conjuntos de nivel de \( f_1 \) y \( f_2 \) son:
        
        \[
        C(c,f_1) =
        \begin{dcases}
        \textnormal{Circunferencia centrada en } (0,0) \textnormal{ de radio } c, & \textnormal{si } c>0, \\
        \{(0,0)\}, & \textnormal{si } c=0, \\
        \emptyset, & \textnormal{si } c<0,
        \end{dcases}
        \]
        \[
        C(c,f_2) =
        \begin{dcases}
        \textnormal{Circunferencia centrada en } (0,0) \textnormal{ de radio } -c, & \textnormal{si } c<0, \\
        \{(0,0)\}, & \textnormal{si } c=0, \\
        \emptyset, & \textnormal{si } c>0.
        \end{dcases}
        \]

Por esta razón, se recurre también al uso de cortes transversales. Dado un plano vertical de la forma \( x = c \) o \( y = c \), un \textbf{corte transversal} es la intersección entre la gráfica de la función y dicho plano. Matemáticamente, corresponde a restringir la función \( f(x, y) \) a una recta del dominio, obteniendo así una función de una sola variable:
\[
\text{Si } x = c, \text{ entonces se considera } g(y) = f(c, y); \quad
\text{si } y = c, \text{ se analiza } h(x) = f(x, c)
\]
En el ejemplo 1, tomando el corte en x=0 podemos observar que:
\[
f(0,y)=0^2+y^2 \Rightarrow f(y)=y^2
\]

    \begin{tikzpicture}[scale=1.5]
        % Ejes coordenados
        \draw[->] (-2.5,0) -- (2.5,0) node[right] {\footnotesize $y$};
        \draw[->] (0,-0.5) -- (0,4.5) node[above] {\footnotesize $z$};

        % Marcas en los ejes
        \foreach \x in {-2,-1,1,2}
            \draw (\x,0.1) -- (\x,-0.1) node[below] {\footnotesize \x};
        \foreach \y in {1,2,3,4}
            \draw (0.1,\y) -- (-0.1,\y) node[left] {\footnotesize \y};

        % Gráfica de f(y) = y^2
        \draw[domain=-2:2, smooth, variable=\y, blue, thick] 
            plot ({\y}, {\y*\y});

        % Etiqueta de la curva
        \node[blue] at (2, 1.6) {\footnotesize $z = y^2$};
    \end{tikzpicture}
\end{center}
