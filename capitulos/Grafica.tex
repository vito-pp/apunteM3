

\begin{obs} Si en la definic\'ion  (\ref{recuerdomate1}) cambiemos  $A \subseteq \mathbb{R}$ por $A \subseteq \mathbb{R}^{n}$  obtenemos de manera an\'aloga la siguiente definici\'on.    
\end{obs}



\begin{definition}  \label{def:grafica}
 
 Sea \( f: A \subset \mathbb{R}^n \); se define la \textit{gráfica de $f$ }, y se nota $graf(f)$,  a
 \[
graf(f)=\{(x_1,x_2,...,x_n,x_{n+1})\in \mathbb{R}^{n+1} \backslash (x_1,x_2,...,x_n) \in A \land f(x_1,x_2,...,x_n)=x_{n+1} \}
 \]

\end{definition}
\begin{obs}  Si \( f: A \subset \mathbb{R}^n \) entonces $graf(f) \subset \mathbb{R}^{n+1}.$
\end{obs}

\begin{obs}  Dada la  anterio observaci\'onr y con el objetivo de poder dibujar,  nos detendremos especialmente en el caso de gr\'aficas de  campos  $f: A\subseteq \mathbb{R}^2 \rightarrow \mathbb{R},$ cuya definici\'on se desprende automaticamente  de  (\ref{def:grafica}), o sea, el caso
 \[
graf(f)=\{(x,y,z)\in \mathbb{R}^3 \backslash (x,y)\subseteq A \land z=f(x,y) \}
 \]
\end{obs}
